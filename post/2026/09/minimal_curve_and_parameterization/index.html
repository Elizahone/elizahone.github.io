void openmesh2libigl(MyMesh& mesh,
                     Eigen::MatrixXd& V,
                     Eigen::MatrixXi& F) {
    int n_vertices = mesh.n_vertices();
    int n_faces = mesh.n_faces();

    V.resize(n_vertices, 3);
    F.resize(n_faces, 3);

    std::map<MyMesh::VertexHandle, int> vertex_index;

    int v_idx = 0;
    for (MyMesh::VertexIter it = mesh.vertices_begin(); it != mesh.vertices_end(); it++) {
        const MyMesh::Point& p = mesh.point(*it);
        V(v_idx, 0) = p[0];
        V(v_idx, 1) = p[1];
        V(v_idx, 2) = p[2];

        vertex_index[*it] = v_idx;
        v_idx++;
    }

    int f_idx = 0;
    for (auto it = mesh.faces_begin(); it != mesh.faces_end(); it++) {
        int count = 0;
        for (MyMesh::FaceVertexIter fvit = mesh.fv_iter(*it); fvit.is_valid(); fvit++) {
            F(f_idx, count) = vertex_index[*fvit];
            count++;
        }
        f_idx++;
    }
}


double voronoi_area(MyMesh& mesh, MyMesh::VertexHandle vh) {
    double area = 0.0f;
    for (auto vf_it = mesh.vf_iter(vh); vf_it.is_valid(); vf_it++) {
        MyMesh::Point pointTemp[3];
        int k = 0;
        for (auto fv_it = mesh.fv_iter(*vf_it); fv_it.is_valid(); fv_it++) {
            pointTemp[k] = mesh.point(*fv_it);
            if (k != 0 && vh == *fv_it) {
                MyMesh::Point temp;
                temp = pointTemp[0];
                pointTemp[0] = pointTemp[k];
                pointTemp[k] = temp;
            }
            k++;
        }
        double costheta = ((pointTemp[1] - pointTemp[0]) | (pointTemp[2] - pointTemp[0])) /
            (sqrt((pointTemp[1] - pointTemp[0]) | (pointTemp[1] - pointTemp[0])) *
            sqrt((pointTemp[2] - pointTemp[0]) | (pointTemp[2] - pointTemp[0])));
        double cosAlphal = ((pointTemp[0] - pointTemp[1]) | (pointTemp[2] - pointTemp[1])) /
            (sqrt((pointTemp[0] - pointTemp[1]) | (pointTemp[0] - pointTemp[1])) *
            sqrt((pointTemp[2] - pointTemp[1]) | (pointTemp[2] - pointTemp[1])));
        double cosAlpha2 = ((pointTemp[1] - pointTemp[2]) | (pointTemp[0] - pointTemp[2])) /
            (sqrt((pointTemp[1] - pointTemp[2]) | (pointTemp[1] - pointTemp[2])) *
            sqrt((pointTemp[0] - pointTemp[2]) | (pointTemp[0] - pointTemp[2])));
        if (costheta < 0 || cosAlphal < 0 || cosAlpha2 < 0)
        {
            double areaT = 0.5*sqrt(1 - costheta * costheta)*sqrt((pointTemp[1] - pointTemp[0]) |
            (pointTemp[1] - pointTemp[0])) * sqrt((pointTemp[2] - pointTemp[0]) | (pointTemp[2] - pointTemp[0]));
            if (costheta < 0) area += areaT * 0.5;
            else area += areaT * 0.25;
        }
        else
        {
            double cotAlphal, cotAlpha2;
            cotAlphal = cosAlphal / sqrt(1 - cosAlphal * cosAlphal);
            cotAlpha2 = cosAlpha2 / sqrt(1 - cosAlpha2 * cosAlpha2);
            area += 0.125*(((pointTemp[1] - pointTemp[0]) | (pointTemp[1] - pointTemp[0]))*
            cotAlpha2 + ((pointTemp[2] - pointTemp[0]) | (pointTemp[2] - pointTemp[0]))*cotAlphal);
        }
    }
    return area;
}


std::map<MyMesh::VertexHandle, MyMesh::Point> meanCurvature(MyMesh mesh)
{
    //找到一邻域所有顶点，对每个顶点，求其相邻两个面的角的正切值
    std::map<MyMesh::VertexHandle, MyMesh::Point> result;
    for (auto v_it = mesh.vertices_begin();v_it != mesh.vertices_end() ;++v_it)
    {
        MyMesh::Point addij = { 0.0,0.0,0.0 };
        for (auto voh_it = mesh.voh_iter(*v_it);voh_it.is_valid() ;++voh_it)
        {
            MyMesh::HalfedgeHandle et = *voh_it;
            double cot[2];
            for (int i = 0;i < 2;i++)
            {
                OpenMesh::Vec3d t[3];
                t[0] = mesh.point(mesh.from_vertex_handle(et));
                t[1] = mesh.point(mesh.to_vertex_handle(et));
                MyMesh::HalfedgeHandle nexte = mesh.next_halfedge_handle(et);
                t[2] = mesh.point(mesh.to_vertex_handle(nexte));//求t2角的正切
                double cost2 = ((t[0] - t[2]) | (t[1] - t[2])) /
                    (sqrt((t[0] - t[2]) | (t[0] - t[2])) * sqrt((t[1]-t[2])|(t[1]-t[2])));
                cot[i] = cost2 / sqrt(1 - cost2 * cost2);
                et = mesh.opposite_halfedge_handle(et);
            }
            MyMesh::Point vj = mesh.point(mesh.to_vertex_handle(*voh_it)), vi = mesh.point(*v_it);
            addij += (cot[0] + cot[1]) * (vj - vi);
        }
        double mixedArea = voronoi_area(mesh, *v_it);
        result[*v_it] = 0.5 * addij / mixedArea;
    }
    return result;
}

void minimal_curve(MyMesh& mesh, const int interations, float lambda) {
    for (int i = 0; i < interations; i++) {
        auto MCF = meanCurvature(mesh);

        for (auto v_it = mesh.vertices_begin(); v_it != mesh.vertices_end(); v_it++) {
            if (mesh.is_boundary(*v_it)) continue;
            MyMesh::Point new_pos = mesh.point(*v_it) + lambda * MCF[*v_it] / 2;
            mesh.point(*v_it) = new_pos;
        }
        // mesh.update_normals();
    }
}
void global_minimal_curve(MyMesh& mesh) {
    int n = mesh.n_vertices();
    Eigen::SparseMatrix<double> A(n, n);
    Eigen::SparseLU<Eigen::SparseMatrix<double>, Eigen::COLAMDOrdering<int> > solver;
    std::vector<Eigen::Triplet<double>> triplets;
    Eigen::VectorXd b_x(n), b_y(n), b_z(n);

    for (MyMesh::VertexIter v_it = mesh.vertices_begin(); v_it != mesh.vertices_end(); v_it++) {

        int idx = v_it->idx();
        if (mesh.is_boundary(*v_it)) {
            triplets.push_back({ idx, idx, 1.0f });
            b_x[idx] = mesh.point(*v_it)[0];
            b_y[idx] = mesh.point(*v_it)[1];
            b_z[idx] = mesh.point(*v_it)[2];
            continue ;
        }

        double count = 0.0f;
        for (auto voh_it = mesh.voh_iter(*v_it);voh_it.is_valid() ;++voh_it)
        {
            MyMesh::HalfedgeHandle et = *voh_it;
            double cot[2];
            for (int i = 0;i < 2;i++)
            {
                OpenMesh::Vec3d t[3];
                t[0] = mesh.point(mesh.from_vertex_handle(et));
                t[1] = mesh.point(mesh.to_vertex_handle(et));
                MyMesh::HalfedgeHandle nexte = mesh.next_halfedge_handle(et);
                t[2] = mesh.point(mesh.to_vertex_handle(nexte));//求t2角的正切
                double cost2 = ((t[0] - t[2]) | (t[1] - t[2])) /
                    (sqrt((t[0] - t[2]) | (t[0] - t[2])) * sqrt((t[1]-t[2])|(t[1]-t[2])));
                cot[i] = cost2 / sqrt(1 - cost2 * cost2);
                et = mesh.opposite_halfedge_handle(et);
            }
            double w = cot[0] + cot[1];
            triplets.push_back({ idx, voh_it->to().idx(), -w });
            count += w;
        }
        triplets.push_back({ idx, idx, count });
        b_x[idx] = 0;
        b_y[idx] = 0;
        b_z[idx] = 0;
    }

    A.setFromTriplets(triplets.begin(), triplets.end());
    solver.analyzePattern(A);
    solver.factorize(A);
    Eigen::VectorXd x = solver.solve(b_x);
    Eigen::VectorXd y = solver.solve(b_y);
    Eigen::VectorXd z = solver.solve(b_z);


    for (MyMesh::VertexIter v_it = mesh.vertices_begin(); v_it != mesh.vertices_end(); v_it++) {
        int idx = v_it->idx();
        // if (!mesh.is_boundary(*v_it))
            mesh.set_point(*v_it, { x[idx], y[idx], z[idx] });
    }
}

double get_weight_cotangent(MyMesh::VertexOHalfedgeIter& voh_it) {
    MyMesh::HalfedgeHandle et = *voh_it;
    double cot[2];
    for (int i = 0;i < 2;i++)
    {
        OpenMesh::Vec3d t[3];
        t[0] = mesh.point(mesh.from_vertex_handle(et));
        t[1] = mesh.point(mesh.to_vertex_handle(et));
        MyMesh::HalfedgeHandle nexte = mesh.next_halfedge_handle(et);
        t[2] = mesh.point(mesh.to_vertex_handle(nexte));//求t2角的正切
        double cost2 = ((t[0] - t[2]) | (t[1] - t[2])) /
            (sqrt((t[0] - t[2]) | (t[0] - t[2])) * sqrt((t[1]-t[2])|(t[1]-t[2])));
        cot[i] = cost2 / sqrt(1 - cost2 * cost2);
        et = mesh.opposite_halfedge_handle(et);
    }
    return cot[0] + cot[1];
}

MyMesh parametrization(MyMesh mesh) {
    const double PI = 3.141592653;
    int n_vertices = mesh.n_vertices();
    Eigen::SparseMatrix<double> A(n_vertices, n_vertices);
    Eigen::VectorXd b_u(n_vertices), b_v(n_vertices);
    Eigen::SparseLU<Eigen::SparseMatrix<double>, Eigen::COLAMDOrdering<int> > solver;
    b_u.setZero(); b_v.setZero();
    std::vector<Eigen::Triplet<double>> triplets;
    std::vector<int> boundary_index;
    for (MyMesh::HalfedgeIter he_it = mesh.halfedges_begin(); he_it != mesh.halfedges_end(); he_it++) {
        if (mesh.is_boundary(*he_it)) {
            int idx = mesh.to_vertex_handle(*he_it).idx();
            boundary_index.push_back(idx);
            MyMesh::HalfedgeHandle he_it_loop = he_it->next();
            while (he_it_loop != *he_it) {
                boundary_index.push_back(mesh.to_vertex_handle(he_it_loop).idx());
                he_it_loop = mesh.next_halfedge_handle(he_it_loop);
            }
            break;
        }
    }
    for (int i = 0; i < (int)boundary_index.size(); i++) {
        double angle = 2 * PI * i / (float)boundary_index.size();
        b_u[boundary_index[i]] = -std::cos(angle);
        b_v[boundary_index[i]] = -std::sin(angle);
    }

    for (MyMesh::VertexIter v_it = mesh.vertices_begin(); v_it != mesh.vertices_end(); v_it++) {
        int row = v_it->idx();
        if (mesh.is_boundary(*v_it)) {
            triplets.push_back( { row, row, 1.0f } );
            continue;
        }

        // N-1 ring
        double total_weight = 0.0f;
        for (auto voh_it = mesh.voh_iter(*v_it);voh_it.is_valid() ;++voh_it)
        {
            double w = get_weight_cotangent(voh_it);
            if (mesh.is_boundary(mesh.to_vertex_handle(*voh_it))) {
                b_u[row] += w * b_u[mesh.to_vertex_handle(*voh_it).idx()];
                b_v[row] += w * b_v[mesh.to_vertex_handle(*voh_it).idx()];
            } else {
                triplets.push_back({ row, voh_it->to().idx(), -w });
            }
            total_weight += w;
        }
        triplets.push_back({ row, row, total_weight });
    }
    A.setFromTriplets(triplets.begin(), triplets.end());
    solver.analyzePattern(A);
    solver.factorize(A);
    Eigen::VectorXd u = solver.solve(b_u);
    Eigen::VectorXd v = solver.solve(b_v);
    for (MyMesh::VertexIter v_it = mesh.vertices_begin(); v_it != mesh.vertices_end(); v_it++) {
        int idx = v_it->idx();
        // if (!mesh.is_boundary(*v_it))
            mesh.set_point(*v_it, { u[idx], v[idx], 0 });
    }
    return mesh;
}

#include "simplify.h"
#include "Eigen/src/Core/Matrix.h"
#include <queue>

void simplify(MyMesh& mesh, double ratio)
{
    mesh.request_face_normals();
    mesh.update_face_normals();
    int it_num = (1.0f - ratio) * mesh.n_vertices();

    auto Q = OpenMesh::makeTemporaryProperty<MyMesh::VertexHandle, Eigen::Matrix4d>(mesh);
    // since we will update / delete vertex during processing, so I set a timestamp for each vertex
    // to indicate the status of vertex.
    auto timestamp = OpenMesh::makeTemporaryProperty<MyMesh::VertexHandle, int>(mesh);
    auto p = OpenMesh::makeTemporaryProperty<MyMesh::FaceHandle, Eigen::Vector4d>(mesh);
    auto v = OpenMesh::makeTemporaryProperty<MyMesh::VertexHandle, Eigen::Vector4d>(mesh);

    // 1. initiliza Q matrix
    for (auto f_it = mesh.faces_begin(); f_it != mesh.faces_end(); f_it++) {
        OpenMesh::VectorT<float, 3> normal = mesh.normal(*f_it);
        normal.normalize();
        double a = normal[0], b = normal[1], c = normal[2];
        auto pp = mesh.point(f_it->halfedge().to());
        double d = - (a * pp[0] + b * pp[1] + c * pp[2]);

        p[*f_it] = {a, b, c, d};
    }
    for (auto v_it = mesh.vertices_begin(); v_it != mesh.vertices_end(); v_it++) {
        Eigen::Matrix4d tmp; tmp.setZero();
        for (auto vf_it = mesh.vf_iter(*v_it); vf_it.is_valid(); vf_it++) {
            tmp += p[*vf_it] * p[*vf_it].transpose();
        }
        Q[*v_it] = tmp;
        v[*v_it][0] = mesh.point(*v_it)[0];
        v[*v_it][1] = mesh.point(*v_it)[1];
        v[*v_it][2] = mesh.point(*v_it)[2];
        v[*v_it][3] = 1.0f;
        timestamp[*v_it] = 0;
    }

    // 2. push into heap
    std::priority_queue<V_Pair> q;
    for (auto e_it = mesh.edges_begin(); e_it != mesh.edges_end(); e_it++) {
        Eigen::Matrix4d Q_new = Q[e_it->v0()] + Q[e_it->v1()];
        Eigen::Matrix4d tQ = Q_new;
        Eigen::Vector4d b = { 0.0, 0.0, 0.0, 1.0 };
        tQ(3, 0) = 0.0, tQ(3, 1) = 0.0, tQ(3, 2) = 0.0, tQ(3, 3) = 1.0;
        Eigen::FullPivLU<Eigen::Matrix4d> lu(tQ);
        Eigen::Vector4d nv;
        if (lu.isInvertible()) {
            nv = tQ.inverse() * b;
        } else {
            nv = (v[e_it->v0()] + v[e_it->v1()]) / 2.0f;
        }

        V_Pair vp;
        vp.cost = nv.transpose() * Q_new * nv;
        vp.he = e_it->halfedge(0);
        vp.newQ = Q_new;
        vp.new_pos = { nv[0], nv[1], nv[2] };
        vp.v_from = e_it->halfedge(0).from();
        vp.v_to= e_it->halfedge(0).to();
        vp.v_from_timestamp = timestamp[e_it->halfedge(0).from()];
        vp.v_to_timestamp = timestamp[e_it->halfedge(0).to()];
        q.push(vp);

    }
    // 3. interatively contract edge;
    mesh.request_vertex_status();
    mesh.request_face_status();
    mesh.request_edge_status();
    int i = 0;
    while (i < it_num) {

        V_Pair vp = q.top();
        q.pop();
        if (mesh.status(vp.v_from).deleted() || mesh.status(vp.v_to).deleted()) continue;
        if (vp.v_from_timestamp != timestamp[vp.v_from] || vp.v_to_timestamp != timestamp[vp.v_to]) {
            continue;
        }
        // contraction
        MyMesh::VertexHandle v_bar;
        if (mesh.is_collapse_ok(vp.he)) {
            mesh.collapse(vp.he);
            v_bar = vp.v_to;
        } else if (mesh.is_collapse_ok(mesh.opposite_halfedge_handle(vp.he))) {
            mesh.collapse(mesh.opposite_halfedge_handle(vp.he));
            v_bar = vp.v_from;
        } else {
            continue ;
        }
        timestamp[vp.v_from]++;
        timestamp[vp.v_to]++;

        mesh.set_point(v_bar, vp.new_pos);
        Q[v_bar] = vp.newQ;
        v[v_bar] = { vp.new_pos[0], vp.new_pos[1], vp.new_pos[2], 1.0 };
        for (auto voh_it = mesh.voh_iter(v_bar); voh_it.is_valid(); voh_it++) {
            auto vj = voh_it->to();
            Eigen::Matrix4d Q_new = Q[v_bar] + Q[vj];
            Eigen::Matrix4d tQ = Q_new;
            Eigen::Vector4d b = { 0.0, 0.0, 0.0, 1.0 };
            tQ(3, 0) = 0.0, tQ(3, 1) = 0.0, tQ(3, 2) = 0.0, tQ(3, 3) = 1.0;
            Eigen::FullPivLU<Eigen::Matrix4d> lu(tQ);
            Eigen::Vector4d nv;
            if (lu.isInvertible()) {
                nv = tQ.inverse() * b;
            } else {
                nv = (v[v_bar] + v[vj]) / 2.0f;
            }
            V_Pair vp_;
            vp_.cost = nv.transpose() * Q_new * nv;
            vp_.he = *voh_it;
            vp_.newQ = Q_new;
            vp_.new_pos = { nv[0], nv[1], nv[2] };
            vp_.v_from = v_bar;
            vp_.v_to= vj;
            vp_.v_from_timestamp = timestamp[v_bar];
            vp_.v_to_timestamp = timestamp[vj];
            q.push(vp_);
        }
        i++;
    }


    mesh.garbage_collection();
}

    double cost;
    MyMesh::Point new_pos;
    Eigen::Matrix4d newQ;
    bool operator< (const V_Pair& other) const {
        return this->cost > other.cost;

};
